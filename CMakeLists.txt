cmake_minimum_required(VERSION 3.25)
project(PelicanExecutor VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel)
qt_add_executable(PelicanExecutor src/main.cpp src/bridge.h src/bridge.cpp)
target_link_libraries(PelicanExecutor PRIVATE Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel)
